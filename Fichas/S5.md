// ===== CalculadoraBilheteira.java (só lógica de negócio, sem I/O) =====
public class CalculadoraBilheteira {
    private static final int    MINIMO_PESSOAS_GRUPO   = 20;    // era "n > 20"
    private static final double FATOR_DESCONTO_GRUPO   = 0.85;  // 15% de desconto
    private static final double DESCONTO_FIXO_SENIOR   = 1.00;  // em euros, sobre o total
    private static final double FATOR_RESIDENTE_PORTO  = 0.50;  // 50% de desconto

    public double calcularPreco(TipoBilhete tipo, int quantidade,
                                boolean temCartaoSenior, boolean residentePorto) {
        double total = tipo.getPrecoUnitario() * quantidade;    // preço base
        total = aplicarDescontoDoTipo(tipo, quantidade, temCartaoSenior, total);
        if (residentePorto) {                                   // regra comum a todos os tipos
            total = total * FATOR_RESIDENTE_PORTO;
        }
        return total;
    }

    private double aplicarDescontoDoTipo(TipoBilhete tipo, int quantidade,
                                         boolean temCartaoSenior, double total) {
        switch (tipo) {
            case GRUPO_ESCOLAR:
                return quantidade > MINIMO_PESSOAS_GRUPO ? total * FATOR_DESCONTO_GRUPO : total;
            case SENIOR:
                return temCartaoSenior ? total - DESCONTO_FIXO_SENIOR : total;
            default:
                return total;                                   // JOVEM_ESTUDANTE: sem desconto
        }
    }
}

        try (Connection c = DriverManager.getConnection(url, user, pass);   // fecha sozinho
             PreparedStatement ps = c.prepareStatement(sql)) {
            ps.setString(1, tipo.name());                       // 1.º parâmetro: nome do enum
            ps.setDouble(2, valor);                             // 2.º parâmetro: valor

        } catch (SQLException e) {

        }
